package com.crs.dashboardService.utils;


import java.io.File;
import java.nio.file.Path;

public class SanitizeUtil {
    public static final String ALLOWED_CHARS = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789_-(). ";
    public static final String ALLOWED_DIGITS = "0123456789";

    public static String sanitizeSegment(String value, String fieldName) {
    if (value == null || value.isEmpty() || value.contains("..")) {
        throw new IllegalArgumentException("Invalid characters in " + fieldName);
    }
    StringBuilder clean = new StringBuilder(value.length());
    for (int i = 0; i < value.length(); i++) {
        char c = value.charAt(i);
        int idx = ALLOWED_CHARS.indexOf(c);
        if (idx < 0) {
            throw new IllegalArgumentException("Invalid character in " + fieldName);
        }
        clean.append(ALLOWED_CHARS.charAt(idx));
    }
    return clean.toString();
    }

    public static File resolveSafeFile(String baseDir, String... segments) {
        Path basePath = new File(baseDir).toPath().toAbsolutePath().normalize();
        Path resolved = basePath;
        for (String segment : segments) {
            String cleanSegment = sanitizeSegment(segment, "path segment");
            resolved = resolved.resolve(cleanSegment).normalize();
        }
        if (!resolved.startsWith(basePath)) {
            throw new SecurityException("Path traversal attempt detected.");
        }
        return resolved.toFile();
    }

    public static int sanitizeInt(Object rawValue, String fieldName) {
        if (rawValue == null) {
            throw new IllegalArgumentException(fieldName + " cannot be null");
        }
        String str = rawValue.toString().trim();
        if (str.isEmpty()) {
            throw new IllegalArgumentException(fieldName + " cannot be empty");
        }
        StringBuilder clean = new StringBuilder(str.length());
        for (int i = 0; i < str.length(); i++) {
            char c = str.charAt(i);
            int idx = ALLOWED_DIGITS.indexOf(c);
            if (idx < 0) {
                throw new IllegalArgumentException("Invalid digit in " + fieldName);
            }
            clean.append(ALLOWED_DIGITS.charAt(idx));
        }
        return Integer.parseInt(clean.toString());
    }


}

    @Override
    @Transactional
    public ResponseEntity<Map<String, Object>> approveRequests(Map<String, Object> payload) {
        Map<String, Object> response = new HashMap<>();
        Map<String, Object> data = (Map<String, Object>) payload.get("data");
        List<Integer> requestIds = (List<Integer>) data.get("requestIds");
        String quarterDate = (String) data.get("quarterDate");
        String userId = String.valueOf(data.get("userId")) ;

        if (requestIds == null || requestIds.isEmpty() || quarterDate == null) {
            response.put("message", "Bad Request: 'requestIds' and 'quarterDate' are required.");
            return new ResponseEntity<>(response, HttpStatus.BAD_REQUEST);
        }

        List<Integer> records = crsAuditStatusRepository.fetchCrsAuditIds(requestIds,quarterDate);
        int successCount = 0;
        try {
            for (Integer reqId : records) {

                List<Object[]> resultList = crsAuditStatusRepository.findRequestDetailsById(String.valueOf(reqId), quarterDate);
                if (resultList == null) continue;
                Object[] details = resultList.get(0);
                String branchCode = String.valueOf(details[0]) ;
                String beforeSts = String.valueOf(details[1]);
                String afterSts = String.valueOf(details[2]);
                String ifcofrFlagForUpdate = "IFCOFR Audited".equalsIgnoreCase(afterSts) ? "Y" : "N";

                String marQuarter = quarterDate.substring(3, 5);

                if ("Non-Audited".equalsIgnoreCase(beforeSts) && "Audited".equalsIgnoreCase(afterSts)) {
                    crsAuditStatusRepository.updateBranchMaster("A", branchCode);
                    crsAuditStatusRepository.updateReportsMasterListAuditable("Y", branchCode, quarterDate);
                    crsAuditStatusRepository.updateReportsStatusAboveThreshold("29", branchCode, quarterDate);
                    if ("03".equals(marQuarter)) {
                        crsAuditStatusRepository.updateLfarBranch("Y", branchCode, quarterDate);
                    }
                } else if ("Non-Audited".equalsIgnoreCase(beforeSts) && "IFCOFR Audited".equalsIgnoreCase(afterSts)) {
                    crsAuditStatusRepository.updateBranchMaster("I", branchCode);
                    crsAuditStatusRepository.updateReportsMasterListAuditable("Y", branchCode, quarterDate);
                    crsAuditStatusRepository.updateReportsStatusAboveThreshold("29", branchCode, quarterDate);
                    //crsAuditStatusRepository.updateIfcofrAudit(ifcofrFlagForUpdate, userId, branchCode, quarterDate);
                    if ("03".equals(marQuarter)) {
                        crsAuditStatusRepository.updateLfarBranch("Y", branchCode, quarterDate);
                    }
                } else if ("Audited".equalsIgnoreCase(beforeSts) && "IFCOFR Audited".equalsIgnoreCase(afterSts)) {
                    //crsAuditStatusRepository.updateIfcofrAudit(ifcofrFlagForUpdate, userId, branchCode, quarterDate);
                    crsAuditStatusRepository.updateBranchMaster("I", branchCode);
                    crsAuditStatusRepository.updateReport1033Status("39", branchCode, quarterDate);
                } else if ("Audited".equalsIgnoreCase(beforeSts) && "Non-Audited".equalsIgnoreCase(afterSts)) {
                    crsAuditStatusRepository.updateBranchMaster("N", branchCode);
                    crsAuditStatusRepository.updateReportsMasterListAuditable("N", branchCode, quarterDate);
                    crsAuditStatusRepository.deleteReportsMasterListByReportId("1033", branchCode, quarterDate);
                    crsAuditStatusRepository.updateReportsStatusAboveThreshold("29", branchCode, quarterDate);
                    if ("03".equals(marQuarter)) {
                        crsAuditStatusRepository.updateLfarBranch("N", branchCode, quarterDate);
                        crsAuditStatusRepository.deleteLfarReportsMasterList(branchCode, quarterDate);
                    }
                } else if ("IFCOFR Audited".equalsIgnoreCase(beforeSts) && "Audited".equalsIgnoreCase(afterSts)) {
                    //crsAuditStatusRepository.updateIfcofrAudit(ifcofrFlagForUpdate, userId, branchCode, quarterDate);
                    crsAuditStatusRepository.updateBranchMaster("A", branchCode);
                    crsAuditStatusRepository.deleteReportsMasterListByReportId("4065", branchCode, quarterDate);
                    crsAuditStatusRepository.updateReport1033Status("39", branchCode, quarterDate);
                } else if ("IFCOFR Audited".equalsIgnoreCase(beforeSts) && "Non-Audited".equalsIgnoreCase(afterSts)) {
                    crsAuditStatusRepository.updateBranchMaster("N", branchCode);
                    crsAuditStatusRepository.updateReportsMasterListAuditable("N", branchCode, quarterDate);
                    //crsAuditStatusRepository.updateIfcofrAudit(ifcofrFlagForUpdate, userId, branchCode, quarterDate);
                    crsAuditStatusRepository.updateReportsStatusAboveThreshold("29", branchCode, quarterDate);
                    crsAuditStatusRepository.deleteReportsMasterListByReportId("4065", branchCode, quarterDate);
                    crsAuditStatusRepository.deleteReportsMasterListByReportId("1033", branchCode, quarterDate);
                    if ("03".equals(marQuarter)) {
                        crsAuditStatusRepository.updateLfarBranch("N", branchCode, quarterDate);
                        crsAuditStatusRepository.deleteLfarReportsMasterList(branchCode, quarterDate);
                    }
                }
                crsAuditStatusRepository.updateRequestStatus(String.valueOf(reqId), "A");
              //  crsAuditStatusRepository.updateCrsRequestStatus(String.valueOf(reqId), "2");
                successCount++;
            }
            for(Integer ids : requestIds) {
                crsAuditStatusRepository.updateCrsRequestStatus(SanitizeUtil.sanitizeSegment(String.valueOf(ids),  "ids"), "2", SanitizeUtil.sanitizeSegment(userId, "userid"));
            }
            response.put("message", successCount + " request(s) approved successfully.");
            return new ResponseEntity<>(response, HttpStatus.OK);
        } catch (Exception e) {
            response.put("message", "An internal error occurred during approval.");
            return new ResponseEntity<>(response, HttpStatus.INTERNAL_SERVER_ERROR);
        }
    }

    @Override
    @Transactional
    public ResponseEntity<Map<String, Object>> rejectRequests(Map<String, Object> payload) {
        Map<String, Object> response = new HashMap<>();
        Map<String, Object> data = (Map<String, Object>) payload.get("data");
        List<Integer> requestIds = (List<Integer>) data.get("requestIds");
        String quarterDate = (String) data.get("quarterDate");
        String userId = String.valueOf(data.get("userId")) ;

        if (requestIds == null || requestIds.isEmpty()) {
            response.put("message", "Bad Request: No request IDs provided.");
            return new ResponseEntity<>(response, HttpStatus.BAD_REQUEST);
        }
        List<Integer> records = crsAuditStatusRepository.fetchCrsAuditIds(requestIds,quarterDate);

        try {
            for (Integer reqId : records) {
                crsAuditStatusRepository.updateRequestStatus(String.valueOf(reqId), "R"); // 'R' for Rejected
            }
            for(Integer ids : requestIds) {
                crsAuditStatusRepository.updateCrsRequestStatus(SanitizeUtil.sanitizeSegment(String.valueOf(ids),  "ids"), "3", SanitizeUtil.sanitizeSegment(userId, "userid")); 
            }
            response.put("message", requestIds.size() + " request(s) rejected successfully.");
            return new ResponseEntity<>(response, HttpStatus.OK);
        } catch (Exception e) {
            response.put("message", "An error occurred during rejection.");
            return new ResponseEntity<>(response, HttpStatus.INTERNAL_SERVER_ERROR);
        }
    }
