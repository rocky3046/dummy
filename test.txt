@Service("LhoManagerService")
public class LhoManagerServiceImpl implements LhoManagerService {

    Logger log = LogManager.getLogger(LhoManagerServiceImpl.class.getName());

    final
    LhoManagerRepository lhoManagerRepository;

    @Autowired
    LfarStatusRepository lfarStatusRepository;

    public LhoManagerServiceImpl(LhoManagerRepository lhoManagerRepository) {
        this.lhoManagerRepository = lhoManagerRepository;
    }

    @Override
    public Map<String, Object> branchList(Map<String, Object> payload) {
        Map<String, Object> user = (Map<String, Object>) payload.get("user");
        Map<String, Object> data = (Map<String, Object>) payload.get("data");
        Map<String, Object> reportList = new HashMap<>();
        String currentQED = user.get("quarterEndDate").toString();
        String circleCode = user.get("circleCode").toString();
        String quarterDate = data.get("quarterDate").toString();
        log.info("cqed: {} qed: {}", currentQED, quarterDate);

        log.info("current quarter date so list from report_submission");
        reportList.put("branchList", lhoManagerRepository.getCurrBranchList(currentQED, circleCode));


        return reportList;
    }

    @Override
    public Map<String, Object> quarterList(Map<String, Object> payload) {
        Map<String, Object> user = (Map<String, Object>) payload.get("user");
        String quarterDate = user.get("quarterEndDate").toString();
        Map<String, Object> quarterList = new HashMap<>();
        List<String> quarter = lhoManagerRepository.getQuarterList();
        quarterList.put("quarterList", quarter);
        return quarterList;
    }

    @Override
    public Map<String, Object> reportList(Map<String, Object> payload) {
        Map<String, Object> result = new HashMap<>();

        try {
            Map<String, Object> user = (Map<String, Object>) payload.get("user");
            Map<String, Object> data = (Map<String, Object>) payload.get("data");

            String quarterDate = (String) data.get("quarterDate");
            log.info("Quarter Date: {}" + quarterDate);

            String branchCode = (String) data.get("branchCode");
            log.info("branch_code: {}" + branchCode);

            List<Map<String, Object>> reportList = lhoManagerRepository.getReportList(quarterDate, branchCode);
            result.put("reportList", reportList);
            result.put("status", true);
            log.info("Report List retrieved :  {}", reportList);

        } catch (Exception e) {
            log.info("Exception occurred while fetching report List to download : " + e.getMessage());
            result.put("reportList", null);
            result.put("status", false);
        }

        return result;
    }

    @Override
    public int rejectReport(Map<String, Object> payload) {
        int result = 0;

        try {
            Map<String, Object> user = (Map<String, Object>) payload.get("user");
            Map<String, Object> data = (Map<String, Object>) payload.get("data");

            String pfNumber = user.get("pf_number").toString();
            String submissionId = String.valueOf(data.get("submissionId"));
            String branchCode = (String) user.get("branch_code");
            String quarterDate = (String) user.get("quarterEndDate");
            String remarks = "Rejected by LHO Manager";
            String rejectStatus = "36";

            int reject = lhoManagerRepository.rejectReport(pfNumber, submissionId);
            if (reject > 0) {
                result = 1;
                insertLfarStatus(user, rejectStatus, submissionId, remarks, data.get("reportId").toString());
                log.info("Lfar Status Updated");
                lhoManagerRepository.updateBrTrackStatusLhoReject(rejectStatus, quarterDate, branchCode);
                log.info("BR Status Updated");
                log.info("reject report successful : {}", reject);
            }
        } catch (Exception e) {
            log.error("Exception occurred in reject report : {}", e.getMessage());
        }

        return result;
    }

    @Override
    public int rejectCompliance(Map<String, Object> payload) {
        int result = 0;
        try {
//            Map<String, Object> user = (Map<String, Object>) payload.get("user");
            Map<String, Object> data = (Map<String, Object>) payload.get("data");

            String submissionId = String.valueOf(data.get("submissionId"));
            String branchCode = (String) data.get("branchCode");
            String quarterDate = (String) data.get("quarterDate");
            String rejectStatus = "36";

            int reject = lhoManagerRepository.rejectCompliance(branchCode, submissionId);
            int rejectComplianceCount = lhoManagerRepository.getComplianceRejectCount(branchCode, quarterDate);
            log.info("rejectComplianceCount : {}", rejectComplianceCount);
            if (reject > 0 && rejectComplianceCount == 6) {
                result = 1;
                lhoManagerRepository.updateBrTrackStatusLhoReject(rejectStatus, quarterDate, branchCode);
                log.info("BR Status Updated");
                log.info("compliance reject successful");
            }else if(reject > 0){
                result = 1;
                log.info("BR Status Not Updated");
                log.info("compliance reject successful");
            }else {
                log.info("compliance reject not successful");
                log.info("BR Status Not Updated");
            }

        } catch (Exception e) {
            log.error("Exception occurred in rejectCompliance report : {}", e.getMessage());
            result = 0;
        }
        return result;
    }


    /**
     * @Category: Private Functions
     */
    private void insertLfarStatus(Map<String, Object> user, String status, String submissionId, String remarks,
                                  String reportId) {
        try {
            LfarStatus sts = new LfarStatus();
            SimpleDateFormat formatter = new SimpleDateFormat("dd/MM/yyyy");

            log.info("Inside insertLfarStatus");
            sts.setBranchCode(user.get("branch_code").toString());
            sts.setCircleCode(user.get("circleCode").toString());
            sts.setRoCode(user.get("roCode").toString());
            sts.setQuarterDate(formatter.parse(user.get("quarterEndDate").toString()));
            sts.setCrsAuditable(user.get("auditStatus").toString());
            sts.setSubmissionIdFk(submissionId);
            sts.setReportId(reportId);
            sts.setCurrentStatus(CommonConstants.getStat(status));
            sts.setStatus(status);
            sts.setRemarks(remarks);
            sts.setLastUpdateDate(LocalDateTime.now());
            sts.setNilFlag("N");
            sts.setUserId(user.get("pf_number").toString());

            log.info("sts : " + sts);

            lfarStatusRepository.save(sts);

        } catch (ParseException e) {
            log.error("Error parsing date {}", e.getMessage());
        }
    }


}
