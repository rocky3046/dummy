# =============================================================================
# create_manifest
# =============================================================================
def create_manifest(spark, filename, manifest_base, etl_date, content):
    """Writes a manifest file to HDFS. Raises RuntimeError if not confirmed."""

    if isinstance(etl_date, str):
        etl_date_str = etl_date
    else:
        etl_date_str = etl_date.strftime("%Y%m%d")

    final_manifest_dir = f"{manifest_base.rstrip('/')}/{etl_date_str}"

    java_import(spark._jvm, "org.apache.hadoop.fs.FileSystem")
    java_import(spark._jvm, "org.apache.hadoop.fs.Path")

    fs   = spark._jvm.FileSystem.get(spark._jsc.hadoopConfiguration())
    Path = spark._jvm.Path

    hdfs_dir = Path(final_manifest_dir)
    if not fs.exists(hdfs_dir):
        fs.mkdirs(hdfs_dir)

    with open(filename, "w") as f:
        f.write(content)

    dest_path = Path(f"{final_manifest_dir}/{filename}")
    fs.copyFromLocalFile(False, True, Path(filename), dest_path)

    if not fs.exists(dest_path):
        raise RuntimeError(f"Manifest NOT written to HDFS: {dest_path}")

    print(f"Manifest successfully created at: {dest_path}")


# =============================================================================
# delta_table_exists
# =============================================================================
def delta_table_exists(spark, path):
    """Returns True if a Delta table exists at the given HDFS path."""
    java_import(spark._jvm, "org.apache.hadoop.fs.Path")
    java_import(spark._jvm, "org.apache.hadoop.fs.FileSystem")

    fs = spark._jvm.FileSystem.get(spark._jsc.hadoopConfiguration())
    delta_log_path = spark._jvm.Path(f"{path}/_delta_log")
    return fs.exists(delta_log_path)


# =============================================================================
# run_pipeline_precheck
# =============================================================================
def run_pipeline_precheck(pipeline, spark, etl_date, run_id, process_run_id):
    """
    log_etl status reference:
      status=1 → Oracle INSERT → must be called ONCE only, never inside a loop
      status=2 → Kafka UPDATE only
      status=3 → Kafka UPDATE only
    """
    pipeline = pipeline.lower()

    logger = setup_logger(
        f"{pipeline.upper()}_PRECHECK",
        etl_date.strftime("%Y-%m-%d"),
        run_id
    )

    if pipeline == "glif":
        log_etl(spark, process_run_id, "precheck_started_glif", 1,
                "Glif Precheck Pipeline Started")
        required_date = [etl_date - timedelta(days=1), etl_date]   # [T-1, T]

    elif pipeline == "difference":
        log_etl(spark, process_run_id, "precheck_started_difference", 1,
                "Difference Precheck Pipeline Started")
        required_date = [etl_date - timedelta(days=1)]              # [T-1]

    else:
        raise ValueError(f"Unknown pipeline: {pipeline}")

    required_date_str = [d.strftime("%d-%m-%Y") for d in required_date]

    # YYYY-MM-DD strings for Spark string comparison
    required_date_str_yyyy = [d.strftime("%Y-%m-%d") for d in required_date]

    logger.info(f"PIPELINE      = {pipeline.upper()}")
    logger.info(f"REQUIRED_DATE = {required_date_str}")

    # ------------------------------------------------------------------
    # STEP 2 — Path config
    # ------------------------------------------------------------------
    cfg = paths_for_read(pipeline, etl_date, required_date_str)
    datalake_path = cfg["PRECHECK_DATALAKE_PATH"]
    manifest_base = cfg["MANIFEST_BASE_PATH"]

    manifest_file = (
        f"{pipeline.upper()}_precheck_successfull"
        f"_{run_id}"
        f"_Manifest.txt"
    )

    logger.info(f"DATALAKE_PATH = {datalake_path}")

    # ------------------------------------------------------------------
    # STEP 3 — Read delta and find missing dates
    # ------------------------------------------------------------------
    log_etl(spark, process_run_id, "source_dir_check", 1, f"checking if source dir : {datalake_path} exists")
    if not create_hdfs_dir(spark, datalake_path, logger=logger):
        log_etl(spark, process_run_id, "source_dir_check", 3, f"Could not create or find path: {datalake_path}")
        spark.stop()
        os._exit(1)
    log_etl(spark, process_run_id, "source_dir_check", 2, f"Source dir : {datalake_path} exists")

    try:
        if not delta_table_exists(spark, datalake_path):
            logger.info("Delta table not found — all dates require migration.")
            missing_dates = required_date

        else:
            if pipeline == "glif":
                # status=1 → Oracle INSERT — called ONCE (not in loop)
                log_etl(spark, process_run_id, "glif_datalake_reading", 1,
                        "GLIF datalake reading started")
                date_column = "BALANCE_DATE"
            else:
                # status=1 → Oracle INSERT — called ONCE (not in loop)
                log_etl(spark, process_run_id, "difference_datalake_reading", 1,
                        "DIFFERENCE datalake reading started")
                date_column = "RECON_DATE"

            delta_df = spark.read.format("delta").load(datalake_path)


            existing_date_strings = set(
                delta_df
                .filter(
                    date_format(col(date_column), "yyyy-MM-dd")
                    .isin(required_date_str_yyyy)
                )
                .select(
                    date_format(col(date_column), "yyyy-MM-dd").alias("dt")
                )
                .distinct()
                .rdd.flatMap(lambda x: x)
                .collect()
            )

            missing_dates = [
                d for d in required_date
                if d.strftime("%Y-%m-%d") not in existing_date_strings
            ]

            logger.info(f"EXISTING_DATES = {list(existing_date_strings)}")
            logger.info(f"MISSING_DATES  = {[d.strftime('%d-%m-%Y') for d in missing_dates]}")

    except Exception:
        logger.error("Failed to read Delta table", exc_info=True)
        if pipeline == "glif":
            logger.error("Error reading GLIF datalake")         
            # status=3 → Kafka only — safe
            log_etl(spark, process_run_id, "glif_datalake_reading", 3,
                    "Error reading GLIF datalake")
        else:
            logger.error("Error reading DIFFERENCE datalake")    
            # status=3 → Kafka only — safe
            log_etl(spark, process_run_id, "difference_datalake_reading", 3,
                    "Error reading DIFFERENCE datalake")
        raise   # raise for BOTH pipelines

    # ------------------------------------------------------------------
    # CASE A — All dates already present → write manifest immediately
    # ------------------------------------------------------------------
    if len(missing_dates) == 0:

        if pipeline == "glif":
            logger.info("Data present in GLIF datalake for all required dates")
            # status=2 → Kafka only — safe
            log_etl(spark, process_run_id, "glif_datalake_reading", 2,
                    "Data present in GLIF datalake for all required dates")

            create_manifest(
                spark, manifest_file, manifest_base, etl_date,
                f"GLIF | DATA_ALREADY_PRESENT | DATE={required_date_str}"
            )
            logger.info("GLIF STATUS = DATA_ALREADY_PRESENT")
            # status=2 → Kafka only — safe
          
            log_etl(spark, process_run_id, "precheck_started_glif", 2,
                    "Glif Precheck Pipeline Completed — Data Already Present")

        elif pipeline == "difference":
            logger.info("Data present in DIFFERENCE datalake for all required dates")
            # status=2 → Kafka only — safe
            log_etl(spark, process_run_id, "difference_datalake_reading", 2,
                    "Data present in DIFFERENCE datalake for all required dates")

            create_manifest(
                spark, manifest_file, manifest_base, etl_date,
                f"DIFFERENCE | DATA_ALREADY_PRESENT | DATE={required_date_str}"
            )
            logger.info("DIFFERENCE STATUS = DATA_ALREADY_PRESENT")
            # status=2 → Kafka only — safe
            
            log_etl(spark, process_run_id, "precheck_started_difference", 2,
                    "Difference Precheck Pipeline Completed — Data Already Present")

        return True

    # ------------------------------------------------------------------
    # CASE B — Some / all dates missing → migrate only missing dates
    # ------------------------------------------------------------------
    try:
        failed_count = 0

        # ── GLIF migration ────────────────────────────────────────────
        if pipeline == "glif":
            logger.info("GLIF DATA NOT PRESENT FOR SOME/ALL DATES | MIGRATION REQUIRED")
            # status=3 → Kafka only — safe
            log_etl(spark, process_run_id, "glif_datalake_reading", 3,
                    "GLIF Data not present — migration required")

           
            log_etl(spark, process_run_id, "glif_migration", 1,
                    "GLIF Migration Started")

            for dt in missing_dates:
                logger.info(f"Running GLIF migration for {dt.strftime('%d-%m-%Y')}")
                success = run_glif_migration(spark, dt, run_id, process_run_id)

                if not success:
                    failed_count += 1
                    logger.error(f"GLIF | No Oracle data for date: {dt.strftime('%d-%m-%Y')}")
                    # status=3 → Kafka only — safe inside loop
                    log_etl(spark, process_run_id, "glif_migration", 3,
                            f"No Oracle data for GLIF date {dt.strftime('%d-%m-%Y')}")
                else:
                    logger.info(f"GLIF | Migration successful for {dt.strftime('%d-%m-%Y')}")

            logger.info(f"GLIF Migration loop done | failed_count = {failed_count}")

            if failed_count > 0:
                # status=3 → Kafka only — safe
                log_etl(spark, process_run_id, "glif_migration", 3,
                        f"GLIF Migration incomplete — {failed_count} date(s) had no Oracle data")
                raise RuntimeError(
                    f"GLIF | {failed_count} date(s) missing in Oracle — manifest NOT written"
                )

            # status=2 → Kafka only — safe, called ONCE after loop
            log_etl(spark, process_run_id, "glif_migration", 2,
                    "GLIF Migration Completed for all dates")
            logger.info("GLIF Migration Completed")

            create_manifest(
                spark, manifest_file, manifest_base, etl_date,
                f"GLIF | DATA_MIGRATED | DATE={required_date_str}"
            )
            logger.info("GLIF STATUS = DATA_MIGRATED")
            # status=2 → Kafka only — safe
            # 
            log_etl(spark, process_run_id, "precheck_started_glif", 2,
                    "Glif Precheck Pipeline Completed — Data Migrated")

        # ── DIFFERENCE migration ──────────────────────────────────────
        elif pipeline == "difference":
            logger.info("DIFFERENCE DATA NOT PRESENT | MIGRATION REQUIRED")
            # status=3 → Kafka only — safe
            log_etl(spark, process_run_id, "difference_datalake_reading", 3,
                    "DIFFERENCE Data not present — migration required")

           
            log_etl(spark, process_run_id, "difference_migration", 1,
                    "DIFFERENCE Migration Started")

            for dt in missing_dates:
                logger.info(f"Running DIFFERENCE migration for {dt.strftime('%d-%m-%Y')}")
                success = run_difference_migration(spark, dt, run_id, process_run_id)

                if not success:
                    failed_count += 1
                    logger.error(
                        f"DIFFERENCE | No Oracle data for date: {dt.strftime('%d-%m-%Y')}")
                    # status=3 → Kafka only — safe inside loop
                    log_etl(spark, process_run_id, "difference_migration", 3,
                            f"No Oracle data for DIFFERENCE date {dt.strftime('%d-%m-%Y')}")
                else:
                    logger.info(
                        f"DIFFERENCE | Migration successful for {dt.strftime('%d-%m-%Y')}")

            logger.info(f"DIFFERENCE Migration loop done | failed_count = {failed_count}")

            # Manifest written ONLY if failed_count == 0
            if failed_count > 0:
                # status=3 → Kafka only — safe
                log_etl(spark, process_run_id, "difference_migration", 3,
                        f"DIFFERENCE Migration incomplete — {failed_count} date(s) had no Oracle data")
                raise RuntimeError(
                    f"DIFFERENCE | {failed_count} date(s) missing in Oracle — manifest NOT written"
                )

            # status=2 → Kafka only — safe, called ONCE after loop
            log_etl(spark, process_run_id, "difference_migration", 2,
                    "DIFFERENCE Migration Completed for all dates")
            logger.info("DIFFERENCE Migration Completed")

            create_manifest(
                spark, manifest_file, manifest_base, etl_date,
                f"DIFFERENCE | DATA_MIGRATED | DATE={required_date_str}"
            )
            logger.info("DIFFERENCE STATUS = DATA_MIGRATED")
            # status=2 → Kafka only — safe
            # — only called here, removed from main to avoid double Kafka event
            log_etl(spark, process_run_id, "precheck_started_difference", 2,
                    "Difference Precheck Pipeline Completed — Data Migrated")

    except Exception:
        logger.error("MIGRATION FAILED", exc_info=True)
        if pipeline == "glif":
            logger.error("Error in GLIF migration")            
            # status=3 → Kafka only — safe
            log_etl(spark, process_run_id, "glif_migration", 3,
                    "Error in GLIF migration")
        else:
            logger.error("Error in DIFFERENCE migration")     
            # status=3 → Kafka only — safe
            log_etl(spark, process_run_id, "difference_migration", 3,
                    "Error in DIFFERENCE migration")
        raise   # Manifest NOT written on any failure


# =============================================================================
# run_currency_precheck_monthend
# =============================================================================
def run_currency_precheck_monthend(spark, etl_date, run_id, process_run_id):
    """
    Validates FLAG=1 currencies have RATE_DATE >= ETL_DATE.
    Returns True/False. Raises on unexpected exceptions.
    """
    logger = setup_logger(
        "CURRENCY_PRECHECK",
        etl_date.strftime("%Y-%m-%d"),
        run_id
    )

    try:
        # status=1 → Oracle INSERT — called ONCE
        log_etl(spark, process_run_id, "precheck_started_currency_monthend", 1,
                "Currency Precheck Pipeline Started")

        oracle = get_oracle_properties()

        query = f"""
        (
            SELECT COUNT(*) AS MISSING_CNT
            FROM CURRENCY_MASTER
            WHERE FLAG = 1
            AND (RATE_DATE IS NULL OR RATE_DATE < DATE '{etl_date.strftime("%Y-%m-%d")}')
        )
        """

        df = (
            spark.read.format("jdbc")
            .option("url",      oracle["url"])
            .option("user",     oracle["user"])
            .option("password", oracle["password"])
            .option("driver",   oracle["driver"])
            .option("dbtable",  query)
            .load()
        )

        missing_cnt = df.collect()[0]["MISSING_CNT"]

        if missing_cnt > 0:
            logger.error(
                f"CURRENCY PRECHECK FAILED: {missing_cnt} currencies "
                f"missing RATE_DATE for ETL_DATE={etl_date}"
            )
            # status=3 → Kafka only — safe
            log_etl(spark, process_run_id, "precheck_started_currency_monthend", 3,
                    "Currency Precheck Failed")
            return False

        logger.info("CURRENCY PRECHECK PASSED")
        # status=2 → Kafka only — safe
        log_etl(spark, process_run_id, "precheck_started_currency_monthend", 2,
                "Currency Precheck Passed")
        return True

    except Exception:
        logger.error("CURRENCY PRECHECK ERROR", exc_info=True)
        # status=3 → Kafka only — safe
        log_etl(spark, process_run_id, "precheck_started_currency_monthend", 3,
                "Error in Currency Precheck Pipeline")
        raise


# =============================================================================
# main
# =============================================================================
def main(args):

    HDFS_BASE  = get_hdfs_base()
    spark      = create_spark_session("DATALAKE_PRECHECK", HDFS_BASE)
    spark.conf.set("spark.hadoop.fs.defaultFS", HDFS_BASE)

    PROCESS_ID = Process.PRECHECK_PIPELINE

    # Dynamic ETL date
    etl_date = get_etl_date(spark)

    # Dynamic run IDs
    process_run_id  = sys.argv[1] if len(sys.argv) > 1 else get_process_run_id(
        spark, PROCESS_ID, etl_date.strftime("%Y-%m-%d"))
    pipeline_run_id = get_run_id(spark, etl_date.strftime("%Y-%m-%d"), PROCESS_ID)

    master_logger = setup_logger(
        "PRECHECK_MASTER",
        etl_date.strftime("%Y-%m-%d"),
        pipeline_run_id
    )

    master_logger.info(f"Precheck ETL Date : {etl_date}")
    master_logger.info("===== PRECHECK STARTED =====")
    # status=1 → Oracle INSERT — called ONCE
    log_etl(spark, process_run_id, "precheck_started", 1,
            "Precheck Pipeline Started")

    # ------------------------------------------------------------------
    # Month-end currency check
    # ------------------------------------------------------------------
    check_monthend = check_MonthEnd()
    if check_monthend:
        master_logger.info("Month-end detected — running Currency Precheck")
        try:
            currency_ok = run_currency_precheck_monthend(
                spark, etl_date, pipeline_run_id, process_run_id)
            if not currency_ok:
                master_logger.error("Currency Precheck FAILED — aborting pipeline")
                # status=3 → Kafka only — safe
                log_etl(spark, process_run_id, "precheck_started", 3,
                        "Aborting — Currency Precheck Failed")
                spark.stop()
                os._exit(1)
        except Exception:
            master_logger.error(
                "Currency Precheck threw an exception — aborting", exc_info=True)
            spark.stop()
            os._exit(1)

    # ------------------------------------------------------------------
    # GLIF and DIFFERENCE pipeline prechecks
    # ------------------------------------------------------------------
    failed_pipeline_count = 0

    for pipeline in ["glif", "difference"]:
        try:
            run_id = get_run_id(spark, etl_date.strftime("%Y-%m-%d"), PROCESS_ID)
            run_pipeline_precheck(pipeline, spark, etl_date, run_id, process_run_id)
            if pipeline == "glif":
                master_logger.info("GLIF Precheck Pipeline Completed")
            else:
                master_logger.info("DIFFERENCE Precheck Pipeline Completed")

        except Exception:
            master_logger.error(f"{pipeline.upper()} PRECHECK FAILED", exc_info=True)
            failed_pipeline_count += 1

            if pipeline == "glif":
                master_logger.error("Error in GLIF Precheck Pipeline")
                # status=3 → Kafka only — safe
                log_etl(spark, process_run_id, "precheck_started_glif", 3,
                        "Error in GLIF Precheck Pipeline")
                log_etl(spark, process_run_id, "precheck_started", 3,
                        "Error in GLIF Precheck Pipeline")
            else:
                master_logger.error("Error in DIFFERENCE Precheck Pipeline")
                # status=3 → Kafka only — safe
                log_etl(spark, process_run_id, "precheck_started_difference", 3,
                        "Error in DIFFERENCE Precheck Pipeline")
                log_etl(spark, process_run_id, "precheck_started", 3,
                        "Error in DIFFERENCE Precheck Pipeline")

    # ------------------------------------------------------------------
    # Final status
    # ------------------------------------------------------------------
    master_logger.info(
        f"===== PRECHECK FINISHED | Failed pipelines = {failed_pipeline_count} ====="
    )

    if failed_pipeline_count == 0:
        # status=2 → Kafka only — safe
        log_etl(spark, process_run_id, "precheck_started", 2,"Precheck Pipeline Completed Successfully")
        master_logger.info("===== PRECHECK COMPLETED SUCCESSFULLY =====")
        spark.stop()
    else:
        # status=3 → Kafka only — safe
        log_etl(spark, process_run_id, "precheck_started", 3,f"Precheck Failed — {failed_pipeline_count} pipeline(s) failed")
        master_logger.error(f"===== PRECHECK FAILED | {failed_pipeline_count} pipeline(s) failed =====")
        spark.stop()
        os._exit(1)

# =============================================================================
if __name__ == "__main__":
    main(sys.argv)
